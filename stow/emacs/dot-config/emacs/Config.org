* Early Init (=early-init.el=)

Emacs reads =early-init.el= before the package system, UI, or the first frame are initialized, so this is the right place for anything that affects *startup speed* or *what the first frame looks like*.

** Garbage collection during startup

The default GC threshold (~800kB) causes constant collections while loading dozens of packages at startup. Raise it temporarily and lower it back down once startup is finished so GC pauses don't show up later during normal editing.

#+begin_src emacs-lisp :tangle early-init.el

  ;; Give startup a large GC budget so package/config loading doesn't
  ;; stall on collections, then put a sane (but still generous) limit
  ;; back once the frame is up.
  (setq gc-cons-threshold most-positive-fixnum
        gc-cons-percentage 0.6)

  (add-hook 'emacs-startup-hook
            (lambda ()
              (setq gc-cons-threshold (* 32 1024 1024) ; 32MB
                    gc-cons-percentage 0.1)))
#+end_src

* Main Init (=init.el=)

#+begin_src emacs-lisp
  ;;; init.el --- Config for my Emacs -*- lexical-binding: t; -*-

  ;;; Commentary:

  ;; This is my personal Emacs configuration.

  ;;; Code:

  ;; Initialize package sources
  (require 'package)

  (setq package-archives '(("melpa" . "https://melpa.org/packages/")
                           ("elpa" . "https://elpa.gnu.org/packages/")
                           ("nongnu" . "https://elpa.nongnu.org/nongnu/")))

  (package-initialize)
  ;; Don't block a daemon on the network at startup; `use-package' with
  ;; :ensure refreshes the archives itself if a package can't be found.
  (unless (or package-archive-contents (daemonp))
    (package-refresh-contents))

  ;; Initialize use-package
  (unless (package-installed-p 'use-package)
    (package-install 'use-package))

  (require 'use-package)
  (setq use-package-always-ensure t)

  (require 'package-vc)
  (setq use-package-verbose t
        package-vc-allow-build-commands t)

  ;; package-vc-install refuses to touch an existing checkout directory,
  ;; which turns any interrupted :vc install (daemon killed mid-clone,
  ;; a prior crash-loop, etc.) into a permanent startup failure on every
  ;; subsequent launch. Wipe a *broken* checkout before it retries; a
  ;; healthy one (git can resolve HEAD) is left alone so upgrades and
  ;; rebuilds don't lose it.
  ;; advice-add (unlike add-function) is documented to handle advising
  ;; a not-yet-loaded autoloaded function correctly, so this doesn't
  ;; need package-vc/vc-git required up front -- the advice attaches
  ;; now and only actually runs the first time a :vc install happens.
  (advice-add 'package-vc--unpack :before
              (lambda (pkg-desc &rest _)
                (let ((dir (package-desc-dir pkg-desc)))
                  (when (and dir (file-directory-p dir)
                             (not (zerop (call-process
                                          "git" nil nil nil "-C" dir
                                          "rev-parse" "--verify" "HEAD"))))
                    (delete-directory dir t)))))

  ;; Shallow-clone :vc packages (full history is never needed for an
  ;; installed dependency). Scoped to package-user-dir so this doesn't
  ;; affect unrelated git-clone calls (e.g. from Magit) elsewhere.
  (advice-add 'vc-git-clone :around
              (lambda (orig remote directory rev)
                (if (not (file-in-directory-p directory package-user-dir))
                    (funcall orig remote directory rev)
                  (if (null rev)
                      (vc-git--out-ok "clone" "--depth" "1" remote directory)
                    (or (ignore-errors
                          (vc-git--out-ok "clone" "--depth" "1" "--branch" rev
                                          "--single-branch" remote directory))
                        (progn
                          (vc-git--out-ok "clone" remote directory)
                          (let ((default-directory directory))
                            (vc-git--out-ok "checkout" rev)))))
                  directory)))

#+end_src

** Module loader

=init.el= only bootstraps the package system (above) and then loads one module per concern from =lisp/=. Each module is required inside =condition-case=, so a single broken module is reported in =*Messages*= instead of aborting the whole startup. The module files are tangled from the =Modules= tree below, one file each.

#+begin_src emacs-lisp
  (add-to-list 'load-path (expand-file-name "lisp" user-emacs-directory))

  (defvar my/modules
    '(init-core init-ui init-completion init-editing init-files init-vcs
                init-prog init-web init-latex init-org init-comms init-apps init-noctalia)
    "Configuration modules, loaded in this order from the lisp/ directory.")

  (dolist (module my/modules)
    (condition-case err
        (require module)
      (error (message "Failed to load %s: %s" module (error-message-string err)))))

  (provide 'init)
    ;;; init.el ends here
#+end_src

*** =no-littering= -- keep =~/.config/emacs= to just this config

Every other package below writes *something* to disk eventually (auto-saves, caches, history files). =no-littering= redirects all of that into =var/= and =etc/= under =user-emacs-directory=, so it has to be loaded *before* anything that picks its own default (scattered) location, including the =recentf= and =save-place= modes enabled in the =emacs= block -- hence it comes first in this module.

#+begin_src emacs-lisp

  (use-package no-littering
    :demand t
    :config
    ;; no-littering's own README recipe: auto-save files still need an
    ;; explicit redirect, everything else it handles on its own.
    ;;
    ;; Regex ".*" below:
    ;;   .   matches any single character (except a newline)
    ;;   *   repeats the item before it zero or more times
    ;;   so ".*" matches every file name, and every auto-save file
    ;;   is redirected into var/auto-save/
    (setq auto-save-file-name-transforms
          `((".*" ,(no-littering-expand-var-file-name "auto-save/") t))))

#+end_src

    ;; `tab-line-format' reserves a screen row whenever it is non-nil, even if
    ;; it renders empty, so the line is hidden by clearing the variable.
    (defvar my/tab-line--updating nil

                   (unless (equal tab-line-format wanted)
                     (setq tab-line-format wanted))))))
           'nomini t))))

    (add-hook 'buffer-list-update-hook #'my/tab-line-update-visibility)
    (add-hook 'tab-line-mode-hook #'my/tab-line-update-visibility)

    (global-tab-line-mode 1))

    :after (nerd-icons tab-line)
    :config

*** Paredit -- structural editing for Lisp

Follows the EmacsWiki ParEdit page: enable it from the Lisp mode hooks (plus the eval-expression minibuffer and IELM), make ElDoc aware of its commands, and add an "electric RETURN".

#+begin_src emacs-lisp

  (use-package paredit
    :commands (enable-paredit-mode paredit-mode)
    ;; `:hook' appends "-hook" to each name.
    :hook ((emacs-lisp-mode
            eval-expression-minibuffer-setup
            ielm-mode
            lisp-mode
            lisp-interaction-mode) . enable-paredit-mode)
    :bind (:map paredit-mode-map
                ;; paredit's M-s (splice) would shadow the consult M-s prefix
                ("M-s" . nil)
                ("M-S" . paredit-splice-sexp))
    :preface
    ;; "Electric RETURN" from the wiki: pressing RET just before a closing
    ;; character opens an extra line, which paredit-close-round regathers.
    ;;
    ;; Regex "[]})\"]":
    ;;   [ ... ]  a bracket expression: match exactly ONE character from the set
    ;;   a ] placed first inside the brackets is taken literally, so the set holds ]
    ;;   }  and  )  are plain literal characters inside brackets
    ;;   \"  is the Elisp string escape for a double quote (not regex syntax)
    ;; `looking-at' only tests the text starting exactly at point, so this
    ;; fires only when the next character is ], }, ) or a double quote.
    (defvar electrify-return-match "[]})\"]"
      "If this regexp matches the text after point, RET does an electric return.")

    (defun electrify-return-if-match (arg)
      "If text after point matches `electrify-return-match', open and indent
  an empty line between point and that text, then insert a newline."
      (interactive "P")
      (let ((case-fold-search nil))
        (when (looking-at electrify-return-match)
          (save-excursion (newline-and-indent)))
        (newline arg)
        (indent-according-to-mode)))
    :init
    ;; The wiki says local-set-key in a mode hook is better than a global
    ;; binding. The minibuffer is left out on purpose, so its own RET
    ;; handling keeps working.
    (dolist (hook '(emacs-lisp-mode-hook
                    lisp-mode-hook
                    lisp-interaction-mode-hook))
      (add-hook hook (lambda ()
                       (local-set-key (kbd "RET") #'electrify-return-if-match))))
    :config
    ;; Make ElDoc refresh the echo area after paredit commands.
    ;; The argument is a plain prefix string, not a regex.
    (require 'eldoc)
    (eldoc-add-command-completions "paredit-")
    ;; Avoid double-inserted closers, since electric-pair-mode is global.
    (add-hook 'paredit-mode-hook (lambda () (electric-pair-local-mode -1))))

*** =Ibuffer= Settings

Regexes used in the groups below:

- ="^\\*newsticker"= -- ^ anchors to the start of the buffer name, \\* is a literal asterisk, newsticker is literal text: buffers named like =*newsticker...=.

             " "

*** =ERC=

Note that =ERC= uses =pass= authentication integration (=auth-source-pass-enable=, in the core module). Make sure that is enabled.

#+begin_src emacs-lisp
  (defvar my/erc-soju-host "127.0.0.1"
    "Address of the local soju plaintext listener.")

  (defvar my/erc-soju-port 6667
    "Port of the local soju plaintext listener.")

  (defvar my/erc-soju-user "woodenAllen"
    "Soju account name.")

  (defvar my/erc-soju-networks '("libera")
    "Network names configured in soju (see BouncerServ `network status').")

  (defun my/erc-soju (network)
    "Connect to NETWORK through the local soju bouncer."
    (interactive
     (list (completing-read "Network: " my/erc-soju-networks nil t)))
    (erc :server my/erc-soju-host
         :port my/erc-soju-port
         :nick "woodenAllen"
         :user (format "%s/%s" my/erc-soju-user network)
         :password (auth-source-pass-get 'secret "irc/soju")
         :id (intern network)))

  (defun my/erc-soju-libera ()
    "Connect to Libera.Chat through soju."
    (interactive)
    (my/erc-soju "libera"))

  (use-package erc
    :ensure nil
    :bind (("C-c e l" . my/erc-soju-libera)
           ("C-c e c" . my/erc-soju))
    :custom
    ;; Retry forever: the bouncer is local and comes back within seconds.
    (erc-server-reconnect-attempts t)
    (erc-server-reconnect-timeout 10)
    ;; Soju keeps channels joined, so ERC must not manage them.
    (erc-autojoin-channels-alist nil)
    (erc-fill-function #'erc-fill-wrap)
    (erc-fill-column 100)
    (erc-hide-list '("JOIN" "PART" "QUIT"))
    (erc-track-shorten-start 8)
    :config
    (dolist (module '(nicks fill-wrap scrolltobottom))
      (add-to-list 'erc-modules module))
    (erc-update-modules))
#+end_src
